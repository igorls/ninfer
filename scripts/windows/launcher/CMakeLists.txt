cmake_minimum_required(VERSION 3.28)
project(ninfer_launcher LANGUAGES CXX RC)
find_package(CUDAToolkit REQUIRED)
add_executable(ninfer-launcher WIN32 main.cpp setup.rc)
target_compile_features(ninfer-launcher PRIVATE cxx_std_20)
target_compile_definitions(ninfer-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(ninfer-launcher PRIVATE ../../../third_party ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(ninfer-launcher PRIVATE shell32 ole32 comdlg32 ws2_32 winhttp)
configure_file(../../../apps/windows/app.rc.in app.rc @ONLY)
# app.rc.in uses the repository root, rather than this standalone project's root.
file(READ "${CMAKE_CURRENT_BINARY_DIR}/app.rc" icon_resource)
string(REPLACE "${PROJECT_SOURCE_DIR}/apps/windows" "${PROJECT_SOURCE_DIR}/../../../apps/windows" icon_resource "${icon_resource}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app.rc" "${icon_resource}")
target_sources(ninfer-launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
